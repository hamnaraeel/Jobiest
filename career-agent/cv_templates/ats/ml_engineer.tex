% Compiled with plain pdflatex -- these two lines are what make the PDF's
% text layer map back to real Unicode characters instead of raw font

% \\* (not \\) between the heading text and its rule -- the star form
% forbids a page break at that specific linebreak, so a heading can never
% land at the bottom of one page with its underline orphaned onto the
% next.
\newcommand{\sectiontitle}[1]{%
  \vspace{3pt}
  \noindent{\large\bfseries #1}\\*[-3pt]
  \noindent\rule{\linewidth}{0.6pt}\\[1pt]
}

\newcommand{\subsectiontitle}[1]{%
  \vspace{2pt}
  \noindent\textit{\textbf{#1}}\\[1pt]
}

% Single-column layout throughout -- this is the one ATS-safe use of a
% two-column construct, for aligning a title/company on the left against
% a date range on the right without manual \hfill spacing (which some
% parsers mis-order).
\newcommand{\tworow}[2]{%
  \noindent\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}l r@{}}
    #1 & #2
  \end{tabular*}\\[1pt]%
}

% Tight bullet lists using only plain-LaTeX list primitives (no enumitem
% dependency, which this TeX installation doesn't have and can't self-
% install) -- \tightlist for normal bullets, \tightlistzero for
% zero-itemsep lists like Certifications.
\newenvironment{tightlist}{%
  \begin{itemize}
  \setlength{\itemsep}{1pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}

\newenvironment{tightlistzero}{%
  \begin{itemize}
  \setlength{\itemsep}{0pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}
