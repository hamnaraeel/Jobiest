% Job/project entry titles (role, company, project name) render at
% 10.5pt bold -- one notch above the 10pt body text, distinct from both
% body text and the larger section headings below.
\newcommand{\entrytitle}[1]{{\fontsize{10.5}{12.6}\selectfont\bfseries #1}}

% Section heading: 13pt bold, with its own rule. \\* (not \\) between the
% heading text and the rule forbids a page break at that specific
% linebreak, so a heading can never land at the bottom of one page with
% its rule orphaned onto the next. The rule ends with \par (a genuine
% paragraph break, in vertical mode) rather than another \\ -- whatever
% follows (a plain skill-category line, a summary paragraph, a project's
% title line) is then a fresh paragraph that starts its own \noindent cleanly,
% instead of being a manually-broken continuation of the heading's own
% paragraph, which is what was producing a stray first-line indent
% before non-list content. \nopagebreak sits right after that \par, its
% correct usage point, to discourage (not guarantee -- plain pdflatex
% has no stronger primitive) breaking right after the heading. 4pt
% before the heading, 4pt after the rule before the first entry -- the
% tight end of the spec's 4-7pt target.
\newcommand{\sectiontitle}[1]{%
  \par\vspace{4pt}
  \noindent{\fontsize{13}{15}\selectfont\bfseries #1}\\*[-3pt]
  \noindent\rule{\linewidth}{0.6pt}\par\nopagebreak[3]\vspace{4pt}%
}

\newcommand{\subsectiontitle}[1]{%
  \par\vspace{3pt}
  \noindent\textit{\textbf{#1}}\par\vspace{3pt}%
}

% Single-column layout throughout -- this is the one ATS-safe use of a
% two-column construct, for aligning a title/company on the left against
% a date range on the right without manual \hfill spacing (which some
% parsers mis-order). A tabular* row carries noticeably more built-in
% vertical padding than a normal text line, so the linebreak after it
% pulls back with a negative space to net out to ~3pt before the next
% line -- the tight end of the spec's 3-5pt title-to-first-bullet target
% (distinct from the ~1pt between bullets, and the ~8pt between one
% entry and the next, both set where entries are joined in
% cv_render_service.py). \nopagebreak softly discourages splitting the
% title from its first bullet across a page.
\newcommand{\tworow}[2]{%
  \noindent\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}l r@{}}
    #1 & #2
  \end{tabular*}\\[-3pt]\nopagebreak[3]%
}

% Tight bullet lists using only plain-LaTeX list primitives (no enumitem
% dependency, which this TeX installation doesn't have and can't self-
% install) -- \tightlist for normal bullets, \tightlistzero for
% zero-itemsep lists like Certifications. leftmargin/labelsep/labelwidth
% are set explicitly so the bullet glyph sits ~0.22in from the margin
% and bullet text starts ~0.42in -- the same indentation everywhere a
% bullet appears (job history, projects, achievements, certifications),
% rather than whatever \itemize's default happens to compute.
\newenvironment{tightlist}{%
  \begin{itemize}
  \setlength{\leftmargin}{0.42in}
  \setlength{\labelsep}{0.08in}
  \setlength{\labelwidth}{0.12in}
  \setlength{\itemsep}{1pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}

\newenvironment{tightlistzero}{%
  \begin{itemize}
  \setlength{\leftmargin}{0.42in}
  \setlength{\labelsep}{0.08in}
  \setlength{\labelwidth}{0.12in}
  \setlength{\itemsep}{1pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}
