% match the candidate's own master resume, which was itself set in a
% plain serif LaTeX template. No \familydefault override to sans here,
% unlike earlier drafts of this template -- this one intentionally
% mirrors the master document's actual look, not a generic ATS-Arial
% convention.

% Compiled with plain pdflatex -- these two lines are what make the PDF's
% text layer map back to real Unicode characters instead of raw font
% glyph indices, which is what lets pdftotext (and every ATS parser)
% extract clean, correctly-spaced text.
\input{glyphtounicode}
\pdfgentounicode=1

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{2pt}

% Section heading: bold, one notch above body size, with a full-width
% rule immediately beneath it and almost no gap -- the master resume's
% own compact heading style, not a large/spaced-out ATS-template one.
% The rule ends with \par (a genuine paragraph break, in vertical mode)
% so whatever plain-text content follows (a summary paragraph, a skill
% line) starts a fresh paragraph and picks up \noindent correctly,
% instead of being read as a manual-linebreak continuation of the
% heading's own paragraph. \\* between the heading text and the rule
% forbids a page break at that specific point, so the two can never be
% split across pages; \nopagebreak right after the \par discourages
% (LaTeX has no hard guarantee without extra packages) breaking between
% the heading and its first line of content.
\newcommand{\sectiontitle}[1]{%
  \vspace{8pt}
  \noindent{\fontsize{12}{14}\selectfont\bfseries #1}\\*[1pt]
  \noindent\rule{\linewidth}{0.5pt}\par\nopagebreak[3]\vspace{4pt}%
}

\newcommand{\subsectiontitle}[1]{%
  \par\vspace{3pt}
  \noindent\textit{\textbf{#1}}\par\vspace{3pt}%
}

% A single two-column row (left-aligned label, right-aligned value) --
% used both for a single title/date line and, stacked twice with
% \rowbreak between them, for the master resume's two-line entry header
% (company/location, then role/date). No trailing spacing is built in
% here; callers compose the gaps explicitly with \rowbreak so the same
% macro works whether one or two rows precede the bullets.
\newcommand{\tworow}[2]{%
  \noindent\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}l r@{}}
    #1 & #2
  \end{tabular*}%
}

% A tabular* row carries noticeably more built-in vertical padding than
% a normal text line, so this pulls back with a negative space to net
% out to ~3pt before whatever comes next (another \tworow, or the first
% bullet) -- consistent everywhere a \tworow is used.
\newcommand{\rowbreak}{\\[-3pt]\nopagebreak[3]}

% Tight bullet lists using only plain-LaTeX list primitives (no enumitem
% dependency, which this TeX installation doesn't have and can't
% self-install) -- \tightlist for normal bullets, \tightlistzero for
% zero-itemsep lists like Certifications.
\newenvironment{tightlist}{%
  \begin{itemize}
  \setlength{\itemsep}{1pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}

\newenvironment{tightlistzero}{%
  \begin{itemize}
  \setlength{\itemsep}{1pt}
  \setlength{\parskip}{0pt}
  \setlength{\topsep}{1pt}
}{\end{itemize}}
